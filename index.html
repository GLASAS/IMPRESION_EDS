sintaxis_valida(email: str) -> bool: return bool(EMAIL_REGEX.match(email.strip())) def obtener_mx_reales(dominio: str) -> list[str]: """Devuelve solo hosts MX reales; si no hay MX, devuelve lista resolver.timeout = 2.0 resolver.lifetime = 3.0 respuestas = resolver.resolve(dominio, "MX") ordenados = sorted(respuestas, key=lambda r: r.preference) hosts = [str(r.exchange).rstrip(".").lower() for r in ordenados if str(r.exchange).strip()] if hosts: return hosts except Exception: pass # 2) Respaldo con nslookup (soporta inglés y español) try: proc = subprocess.run( ["nslookup", "-type=MX", dominio], capture_output=True, text=True, timeout=3, ) hosts_ns: list[str] = [] for linea in (proc.stdout or "").splitlines(): l = linea.strip().lower() if "mail exchanger" in l or "intercambiador de correo" in l or "mx preference" in l: # Extrae el último token (el nombre del servidor MX) partes = linea.replace("=", " ").replace(",", " ").split() for token in reversed(partes): host_limpio = token.strip(".").lower() if "." in host_limpio and not host_limpio.replace(".", "").isdigit(): hosts_ns.append(host_limpio) break return hosts_ns except Exception: return [] def PROBAR_SMTP_ACTIVO: return "no_verificable_smtp", "Verificación SMTP desactivada" ultimo_motivo = "Puerto 25 bloqueado o sin respuesta SMTP" for host in mx_hosts[:2]: try: with smtplib.SMTP(host=host, port=25, timeout=3.0) as server: server.ehlo_or_helo_if_needed() code_from, _ = server.mail(remitente) if code_from not in (250, 251): ultimo_motivo = f"{host} rechazó MAIL FROM ({code_from})" continue code_rcpt, msg_rcpt = server.rcpt(email) msg = msg_rcpt.decode("utf-8", errors="replace") if isinstance(msg_rcpt, bytes) else str(msg_rcpt) if code_rcpt in (250, 251): return "activo_smtp", f"{host} aceptó destinatario ({code_rcpt})" if code_rcpt in (550, 551, 552, 553, 554): return "inexistente_smtp", f"{host} rechazó buzón ({code_rcpt}: {msg.strip()})" ultimo_motivo = f"{host} respondió {code_rcpt}" except ( smtplib.SMTPServerDisconnected, smtplib.SMTPConnectError, smtplib.SMTPHeloError, TimeoutError, socket.error, OSError, ) as exc: ultimo_motivo = f"{host} ({type(exc).__name__})" continue return "no_verificable_smtp", ultimo_motivo def normalizar_whatsapp(valor: object, respaldo_tel: object = None) -> str: origen = valor if (valor is not None and return "sin_whatsapp", "Sin número" if not TELEFONO_E164_REGEX.match(numero): return "numero_invalido", f"Formato no válido ({numero})" if api_url and api_key: try: resp = requests.post( api_url, https://wa.me/{numero}" def procesar_excel(ruta_entrada: str, nombre_hoja: str = "Hoja1") -> str: t_inicio = time.time() log(f"Cargando workbook desde: {ruta_entrada}") wb = openpyxl.load_workbook(ruta_entrada) if nombre_hoja not in wb.sheetnames: raise ValueError(f"No existe la hoja '{nombre_hoja}'. Hojas disponibles: {wb.sheetnames}") ws = wb[nombre_hoja] total_filas = ws.max_row log(f"Workbook cargado. t_inicio:.1f} s.") return ruta_salida if __name__ == "__main__": try: if len(sys.argv) > 1: archivo = sys.argv[1] else: log("Abriendo selector de archivo...") archivo = seleccionar_archivo_excel() if not archivo: log("No se seleccionó ningún archivo.") sys.exit(0) salida = procesar_excel(archivo, nombre_hoja="Hoja1") log(f"Archivo generado: {salida}") try: root = tk.Tk() root.withdraw() root.wm_attributes("-topmost", 1) messagebox.showinfo("Listo", f"Archivo generado:\n{salida}") root.destroy() except Exception: pass except Exception as exc: log(f"ERROR: {type(exc).__name__}: {exc}") input("Presiona Enter para cerrar...")
